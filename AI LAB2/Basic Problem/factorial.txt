number=int(input("Enter any number:"))
def cal_factorial(number):
    factorial=1
    if number==0 or number==1:
        return 1
    for i in range(1,number+1):
        factorial=factorial*i
    return factorial
    
output=cal_factorial(number)
print('Factorial of number',number,'is:',output)
